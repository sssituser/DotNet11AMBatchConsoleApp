1.	Reactanular Array(Multidimensional array)
		An array with rows and cols can be called Rectangular array.
		Internally arrays can be stored sequentially.

		Syx:
			DataType [ , ] refvariable = new DataType[rows,cols];
			Example :

			int[ ] a = new int[3] ----> Singledimensional array
			a[0]
			a[1]
			a[2]
			int [ , ]  array = new int[2,3]; ---> Rectangular Array
			or Multidimensional Array.
			a[0,0]
			a[0,1]
			a[1,0]
			a[1,1]

			int[ , ] a = new int[2,3]{{11,56,78},{90,98,97}};
				or
			int[ , ] a = {{11,56,78},{90,98,97}};

			a[0,0]=>11
			a[0,1]=>56
			a[0,2]=>78

			a[1,0]=>90
			a[1,1]=>91
			a[1,2]=>97
